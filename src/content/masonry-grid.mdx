# Masonry Grid

A responsive masonry-style image grid with hover animations and lightbox support.

<ComponentPreview
  component={
    <div className="relative w-full rounded-lg overflow-hidden bg-black">
      <PreviewVideo showControls={false} src="https://cdn.obsidianui.dev/demos/masonry-grid.mp4" label="Masonry Grid demo" className="w-full aspect-video" />
    </div>
  }
  code={`import { MasonryGrid } from '@/components/block/masonry-grid'

export function Demo() {
  return (
    <MasonryGrid
      items={[{ image: "/image1.jpg", title: "Mountains", description: "A view from the summit" }, { image: "/image2.jpg", title: "Coast", description: "The shoreline at sunset" }]}
    />
  )
}`}
/>

    <CodeBlock code="npm install motion clsx tailwind-merge" />

    <CodeBlock language="tsx" code={`import { ClassValue, clsx } from 'clsx';
import { twMerge } from 'tailwind-merge';
