# Flip Scroll

A scroll-triggered flip animation where elements rotate and transform as you scroll. Creates a dynamic, engaging scroll experience.

<ComponentPreview
  component={
    <PreviewVideo showControls={false} src="https://cdn.obsidianui.dev/demos/flip-scroll.mp4" label="Flip Scroll demo" className="w-full h-[400px] rounded-lg" />
  }
  code={`import FlipScroll from '@/components/block/flip-scroll'

export function Demo() {
  const items = [
    { image: '/images/one.jpg' },
    { image: '/images/two.jpg' },
    { image: '/images/three.jpg' },
    { image: '/images/four.jpg' },
  ]
  
  return <div className="h-[600px]"><FlipScroll items={items} /></div>
}`}
/>
